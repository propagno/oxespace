import { execFileSync } from 'node:child_process'
// access would let it through? Calls the real evaluatePermission / fullAccessPolicy of the checked-out branch.
import { mkdirSync, mkdtempSync, realpathSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { evaluatePermission, fullAccessPolicy } from '../../../electron/main/services/conversation/kernel/permission-policy.ts'

const root = realpathSync.native(mkdtempSync(join(tmpdir(), 'hml-pp208-')))
mkdirSync(join(root, '.oxespace')); writeFileSync(join(root, '.mcp.json'), '{}'); writeFileSync(join(root, 'AGENTS.md'), 'x: 1\n')
const allowCmd = ['ls', 'grep', '7z', 'Expand-Archive', 'gzip', 'rsync', 'pushd', 'GIT_DIR=.git', 'echo', 'printf', 'python', 'python3', 'node', 'git', 'curl', 'cd', 'tar', 'unzip', 'cp', 'mv', 'sh', 'bash', 'pwsh', 'powershell', 'Set-Content', 'Out-File', 'tee', 'sed', 'cat', 'jq', 'yq', 'npm', 'npx']
const rules = [
  ...allowCmd.map((pattern, i) => ({ id: `c${i}`, projectRoot: root, kind: 'command' as const, pattern, decision: 'allow' as const, createdAt: 0 })),
  { id: 'p1', projectRoot: root, kind: 'path' as const, pattern: '**', decision: 'allow' as const, createdAt: 0 },
  { id: 'p2', projectRoot: root, kind: 'path' as const, pattern: '.*', decision: 'allow' as const, createdAt: 0 }
]
const show = (label: string, req: Record<string, unknown>) => {
  const p = evaluatePermission({ kind: 'approval', ...req } as never, root, rules as never, { windows: true, realpath: (x: string) => { try { return realpathSync.native(x) } catch { return x } } } as never)
  const full = fullAccessPolicy({ kind: 'approval', toolName: req.toolName as string } as never, p, 'full')
  const verdict = `${p?.decision ?? 'none'}${p?.guarded ? ' GUARDED' : ''} | full-access: ${full?.decision ?? 'none'}${full?.guarded ? ' GUARDED' : ''}`
  console.log(label.padEnd(58), verdict)
}
