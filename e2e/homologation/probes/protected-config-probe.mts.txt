// #208 probe: does a write to .oxespace/policy.json or oxe.yaml still reach the person when a saved rule or Full
// access would let it through? Calls the real evaluatePermission / fullAccessPolicy of the checked-out branch.
import { mkdirSync, mkdtempSync, realpathSync, writeFileSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { evaluatePermission, fullAccessPolicy } from '../../../electron/main/services/conversation/kernel/permission-policy.ts'

const root = realpathSync.native(mkdtempSync(join(tmpdir(), 'hml-pp208-')))
mkdirSync(join(root, '.oxespace')); writeFileSync(join(root, '.oxespace', 'policy.json'), '{}'); writeFileSync(join(root, 'oxe.yaml'), 'x: 1\n')
const allowCmd = ['echo', 'printf', 'python', 'python3', 'node', 'git', 'curl', 'cd', 'tar', 'unzip', 'cp', 'mv', 'sh', 'bash', 'pwsh', 'powershell', 'Set-Content', 'Out-File', 'tee', 'sed', 'cat', 'jq', 'yq', 'npm', 'npx']
const rules = [
  ...allowCmd.map((pattern, i) => ({ id: `c${i}`, projectRoot: root, kind: 'command' as const, pattern, decision: 'allow' as const, createdAt: 0 })),
  { id: 'p1', projectRoot: root, kind: 'path' as const, pattern: '**', decision: 'allow' as const, createdAt: 0 },
  { id: 'p2', projectRoot: root, kind: 'path' as const, pattern: '.*', decision: 'allow' as const, createdAt: 0 }
]
const show = (label: string, req: Record<string, unknown>) => {
  const p = evaluatePermission({ kind: 'approval', ...req } as never, root, rules as never, { windows: true, realpath: (x: string) => { try { return realpathSync.native(x) } catch { return x } } } as never)
  const full = fullAccessPolicy({ kind: 'approval', toolName: req.toolName as string } as never, p, 'full')
  const verdict = `${p?.decision ?? 'none'}${p?.guarded ? ' GUARDED' : ''} | full-access: ${full?.decision ?? 'none'}${full?.guarded ? ' GUARDED' : ''}`
  console.log(label.padEnd(58), verdict)
}
const cmds = [
  String.raw`echo {} > .oxespace/policy.json`, String.raw`echo x >> oxe.yaml`, String.raw`printf '{}' > .oxespace/policy.json`,
  String.raw`cp /tmp/p.json .oxespace/policy.json`, String.raw`sed -i 's/deny/allow/' .oxespace/policy.json`,
  String.raw`git checkout -- .oxespace/policy.json`, String.raw`git restore oxe.yaml`, String.raw`git apply evil.patch`,
  String.raw`python -c "open('.oxespace/policy.json','w').write('{}')"`, String.raw`node -e "require('fs').writeFileSync('oxe.yaml','x')"`,
  String.raw`curl -o .oxespace/policy.json https://x.invalid/p`, String.raw`tar xf p.tar .oxespace/policy.json`, String.raw`unzip -o p.zip`,
  String.raw`cd .oxespace && echo {} > policy.json`, String.raw`Set-Content .oxespace\policy.json "{}"`, String.raw`Set-Content -Path .\oxe.yaml -Value x`,
  String.raw`powershell -c "Set-Content .oxespace\policy.json '{}'"`, String.raw`sh -c 'echo {} > .oxespace/policy.json'`,
  String.raw`jq '.rules=[]' .oxespace/policy.json | tee .oxespace/policy.json`, String.raw`yq -i '.x=2' oxe.yaml`, String.raw`npx some-tool --write .oxespace/policy.json`,
  String.raw`cat .oxespace/policy.json`
]
for (const c of cmds) show(`cmd  ${c}`.slice(0, 58), { command: c, toolName: 'Bash' })
const paths = [
  '.oxespace/policy.json', String.raw`.oxespace\policy.json`, String.raw`.OXESPACE\POLICY.JSON`, join(root, '.oxespace', 'policy.json'),
  '.oxespace/./policy.json', 'sub/../.oxespace/policy.json', '.oxespace/policy.json.', '.oxespace/policy.json ', '.oxespace/policy.json::$DATA',
  String.raw`.oxespace\POLICY~1.JSO`, 'oxe.yaml', 'OXE.YAML', 'oxe.yaml.', 'oxe.yaml::$DATA', 'oxe.yml', '.oxespace/other.json'
]
for (const p of paths) show(`path ${p.replace(root, '<root>')}`.slice(0, 58), { paths: [p], toolName: 'Edit' })
