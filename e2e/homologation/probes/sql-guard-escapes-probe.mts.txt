import { guardSql } from '../../../electron/main/services/oxequery/sql-guard.ts'
const c: Record<string, string> = {
  'PG E-string':String.raw`SELECT E'\''; DELETE FROM users`,
  'PG e-string lowercase': String.raw`select e'\''; drop table users`,
  'PG U&-string (control)': String.raw`SELECT U&'d\0061t'; DELETE FROM users`,
  'MySQL backslash quote': String.raw`SELECT '\''; DELETE FROM users`,
  'MySQL double-quoted': String.raw`SELECT "\""; DELETE FROM users`,
  'SQL Server [bracket]': `SELECT [a';b]; DELETE FROM users`,
  'PG E-string inside UPDATE': String.raw`UPDATE t SET a = E'\'' WHERE id = 1; DELETE FROM users`,
  'control: plain doubled quote': `SELECT ''''; DELETE FROM users`,
}
for (const [k, v] of Object.entries(c)) { const g = guardSql(v) as { kind: string; statements?: number }; console.log(k.padEnd(28), g.kind.padEnd(12), 'statements=' + g.statements, '|', v) }
